cmake_minimum_required(VERSION 3.22)
project(rexlauncher LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Launcher requires Android arm64-v8a")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ADRENOTOOLS "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/libadrenotools")
if(NOT EXISTS "${ADRENOTOOLS}/lib/linkernsbypass/CMakeLists.txt")
    message(FATAL_ERROR "Run python3 tools/fetch_deps.py from project root")
endif()
add_subdirectory("${ADRENOTOOLS}" "${CMAKE_CURRENT_BINARY_DIR}/adrenotools")
add_library(rexlauncher SHARED launcher.cpp xdvdfs.cpp gpu.cpp)
target_compile_options(rexlauncher PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers)
target_link_libraries(rexlauncher PRIVATE adrenotools android log dl)
# Vulkan is deliberately not linked. Resolve ALL Vulkan entrypoints through the
# selected handle's vkGetInstanceProcAddr. A linked renderer must do the same.
target_link_options(rexlauncher PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--no-undefined" "-Wl,--exclude-libs,ALL")
# All hook libraries need 16 KiB ELF segment alignment, too.
foreach(hook hook_impl main_hook file_redirect_hook gsl_alloc_hook)
    target_link_options(${hook} PRIVATE "-Wl,-z,max-page-size=16384")
endforeach()

set(REX_GAME_SOURCE "" CACHE PATH "Android port exposing static rex_android_game target")
if(REX_GAME_SOURCE)
    add_subdirectory("${REX_GAME_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/game")
    if(NOT TARGET rex_android_game)
        message(FATAL_ERROR "Game port must define rex_android_game and rex_game_api_v1()")
    endif()
    target_link_libraries(rexlauncher PRIVATE rex_android_game)
    target_compile_definitions(rexlauncher PRIVATE REX_HAS_GAME=1)
else()
    target_compile_definitions(rexlauncher PRIVATE REX_HAS_GAME=0)
endif()
